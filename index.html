Why YouTube Downloads Cannot Run Exclusively Inside Web Browsers

YouTube enforces strict CORS (Cross-Origin Resource Sharing) headers, encrypted signature tokens (SABR/PoToken), and rate-limiting. Web browsers block frontends from directly downloading audio streams from YouTube servers. To process actual YouTube video links into MP3s, a lightweight backend server running yt-dlp and FFmpeg is mandatory.

Node.js + Express + yt-dlp Backend Code

To make this application completely functional with real YouTube video streams, deploy this Node.js script. It invokes yt-dlp and pipes converted MP3 audio directly back to the client.

1. Setup Server Directory & Dependencies: 
mkdir yt-mp3-backend && cd yt-mp3-backend
npm init -y
npm install express cors yt-dlp-exec fluent-ffmpeg
# Ensure FFmpeg and yt-dlp binary are installed on system path
2. Production Ready Server Implementation (server.js): 
const express = require('express');
const cors = require('cors');
const { exec } = require('child_process');
const ytdlp = require('yt-dlp-exec');

const app = express();
app.use(cors());
app.use(express.json());

// Endpoint 1: Extract Playlist Details
app.get('/api/playlist', async (req, res) => {
    const playlistUrl = req.query.url;
    if (!playlistUrl) return res.status(400).json({ error: 'Playlist URL required' });

    try {
        const output = await ytdlp(playlistUrl, {
            dumpSingleJson: true,
            flatPlaylist: true,
            noWarnings: true
        });

        const tracks = output.entries.map((entry, index) => ({
            id: entry.id,
            title: entry.title,
            channel: entry.uploader || output.uploader || 'YouTube Channel',
            duration: entry.duration ? `${Math.floor(entry.duration/60)}:${String(entry.duration%60).padStart(2,'0')}` : '03:30',
            seconds: entry.duration || 210,
            url: `https://www.youtube.com/watch?v=${entry.id}`
        }));

        res.json({
            title: output.title || 'YouTube Playlist',
            channel: output.uploader || 'YouTube Creator',
            tracks
        });
    } catch (err) {
        res.status(500).json({ error: 'Failed to extract playlist metadata', details: err.message });
    }
});

// Endpoint 2: Stream Live Audio Conversion as MP3
app.get('/api/download-mp3', (req, res) => {
    const videoUrl = req.query.url;
    const quality = req.query.quality || '256';
    const title = req.query.title || 'audio';

    if (!videoUrl) return res.status(400).send('Missing video URL');

    const sanitizedTitle = title.replace(/[/\\?%*:|"<>]/g, '_');
    
    res.header('Content-Type', 'audio/mpeg');
    res.header('Content-Disposition', `attachment; filename="${encodeURIComponent(sanitizedTitle)}.mp3"`);

    // Stream directly through yt-dlp & FFmpeg
    const subprocess = ytdlp.exec(videoUrl, {
        extractAudio: true,
        audioFormat: 'mp3',
        audioQuality: `${quality}k`,
        output: '-' // Pipe stream to stdout
    });

    subprocess.stdout.pipe(res);

    req.on('close', () => {
        subprocess.kill();
    });
});

const PORT = process.env.PORT || 4000;
app.listen(PORT, () => console.log(`Server listening on port ${PORT}`));
